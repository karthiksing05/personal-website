{{- /* Ventures carousel. Every file in data/ventures/ is one card, shown newest first by `date`. To add a venture, drop a new .toml file in that folder (copy any existing one). Fields: title required date required, "YYYY-MM" (the month it was built). Cards from the same month are ordered by filename; give one a day ("YYYY-MM-DD") to control which comes first. hackathon optional, shown as a bubble award optional, shown as a bubble; use a list (["...", "..."]) for several team optional, "with ..." line under the title description markdown results list of markdown strings, shown in the results/ folder link main link (a page on this site or an external URL) link_label text for `link` (defaults to "read more") devpost, video, github, site optional URLs */ -}} {{- $items := slice -}} {{- range $slug, $v := site.Data.ventures -}} {{- /* a month-only date sorts as day 00, and the filename breaks any remaining tie */ -}} {{- $day := $v.date -}} {{- if eq (len $day) 7 }}{{ $day = printf "%s-00" $day }}{{ end -}} {{- $items = $items | append (merge $v (dict "slug" $slug "tag" $v.hackathon "sortkey" (printf "%s|%s" $day $slug))) -}} {{- end -}} {{- $items = sort $items "sortkey" "desc" -}} {{- partial "card-deck.html" (dict "items" $items "section" "ventures" "noun" "venture" "list" "results") -}}
